  <script>
    (function () {
      var cardP1Inputs = document.querySelectorAll('input[name="p1_decision"]');
      var cardP2Inputs = document.querySelectorAll('input[name="p2_decision"]');
      var p2ChangesText = document.getElementById('p2_changes_text');
      var jsonDisplay = document.getElementById('jsonDisplay');
      var copyBtn = document.getElementById('copyBtn');
      var downloadBtn = document.getElementById('downloadBtn');
      var statusMsg = document.getElementById('statusMessage');

      function getSelectedValue(nodeList) {
        for (var i = 0; i < nodeList.length; i++) {
          if (nodeList[i].checked) {
            return nodeList[i].value;
          }
        }
        return null;
      }

      function buildPayload() {
        var p1Choice = getSelectedValue(cardP1Inputs);
        var p2Choice = getSelectedValue(cardP2Inputs);
        var p2Notes = (p2ChangesText && p2ChangesText.value) ? p2ChangesText.value.trim() : "";

        var decisionsComplete = Boolean(p1Choice && p2Choice);
        var selectionsApproveRequestedActions = Boolean(
          p1Choice === "authorize_narrow_repair" && p2Choice === "approve_shown_warning"
        );

        return {
          packet_id: "campaign-2283-main-gates",
          target_pull_request: 2297,
          target_commit: "2d9599c11ea91f303b5067c39fb8ea272f119072",
          evaluated_at_utc: new Date().toISOString(),
          decisions: {
            card_p1_main_protection: {
              authorized_action: p1Choice,
              status: p1Choice ? "decided" : "unselected"
            },
            card_p2_ledger_warning: {
              approved_action: p2Choice,
              presentation_change_request: (p2Choice === "request_changes") ? p2Notes : null,
              status: p2Choice ? "decided" : "unselected"
            }
          },
          decisions_complete: decisionsComplete,
          selections_approve_requested_actions: selectionsApproveRequestedActions,
          requires_lead_revalidation: true
        };
      }

      function updateDisplay() {
        var payload = buildPayload();
        jsonDisplay.value = JSON.stringify(payload, null, 2);
      }

      for (var i = 0; i < cardP1Inputs.length; i++) {
        cardP1Inputs[i].addEventListener('change', updateDisplay);
      }
      for (var j = 0; j < cardP2Inputs.length; j++) {
        cardP2Inputs[j].addEventListener('change', updateDisplay);
      }
      if (p2ChangesText) {
        p2ChangesText.addEventListener('input', updateDisplay);
      }

      copyBtn.addEventListener('click', function () {
        var text = jsonDisplay.value;
        if (navigator.clipboard && navigator.clipboard.writeText) {
          navigator.clipboard.writeText(text).then(function () {
            showNotice("Copied to clipboard!");
          }).catch(function () {
            fallbackCopy(text);
          });
        } else {
          fallbackCopy(text);
        }
      });

      function fallbackCopy(text) {
        jsonDisplay.select();
        try {
          var ok = document.execCommand('copy');
          if (ok) {
            showNotice("Copied to clipboard!");
          } else {
            showNotice("Please select and copy manually.");
          }
        } catch (e) {
          showNotice("Please select and copy manually.");
        }
      }

      downloadBtn.addEventListener('click', function () {
        var text = jsonDisplay.value;
        var blob = new Blob([text], { type: 'application/json' });
        var url = URL.createObjectURL(blob);
        var a = document.createElement('a');
        a.href = url;
        a.download = 'founder-decisions-2283.json';
        document.body.appendChild(a);
        a.click();
        document.body.removeChild(a);
        URL.revokeObjectURL(url);
        showNotice("Downloaded JSON file!");
      });

      function showNotice(msg) {
        statusMsg.textContent = msg;
        setTimeout(function () {
          statusMsg.textContent = "";
        }, 3000);
      }

      // Initial state render (no decisions pre-selected)
      updateDisplay();
    })();
  </script>
